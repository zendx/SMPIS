PART 1 — DATABASE SCHEMA
1.0 Conventions
All tables use id BIGSERIAL PRIMARY KEY unless noted.
All tables include created_at TIMESTAMPTZ DEFAULT now(), updated_at TIMESTAMPTZ DEFAULT now(), deleted_at TIMESTAMPTZ NULL (soft delete) — omitted below for brevity but assumed on every table.
Multi-tenant ready: every table also carries school_id BIGINT NOT NULL REFERENCES schools(id) to support the Phase-4 "multiple schools" scale target — omitted below for brevity but assumed on every table except schools itself.
Money stored as NUMERIC(14,2). Currency stored per-school in schools.currency_code.
Enum-like fields are implemented as VARCHAR with a CHECK constraint (portable) rather than native Postgres ENUM, to simplify future value additions.
1.1 Platform / Identity Tables
CREATE TABLE schools (
    id BIGSERIAL PRIMARY KEY,
    name VARCHAR(255) NOT NULL,
    short_code VARCHAR(20) UNIQUE NOT NULL,
    currency_code VARCHAR(3) DEFAULT 'USD',
    logo_url VARCHAR(500),
    timezone VARCHAR(64) DEFAULT 'UTC',
    is_active BOOLEAN DEFAULT true
);
CREATE TABLE roles (
    id BIGSERIAL PRIMARY KEY,
    school_id BIGINT REFERENCES schools(id),
    name VARCHAR(50) NOT NULL, -- SUPER_ADMIN, PROPRIETOR, PRINCIPAL, VICE_PRINCIPAL,
                                -- ADMISSIONS_OFFICER, TEACHER, FINANCE_OFFICER,
                                -- HR_OFFICER, FACILITIES_MANAGER, PARENT, STUDENT
    description VARCHAR(255)
);
CREATE TABLE permissions (
    id BIGSERIAL PRIMARY KEY,
    code VARCHAR(100) UNIQUE NOT NULL, -- e.g. 'students.read', 'finance.write'
    description VARCHAR(255)
);
CREATE TABLE role_permissions (
    role_id BIGINT REFERENCES roles(id),
    permission_id BIGINT REFERENCES permissions(id),
    PRIMARY KEY (role_id, permission_id)
);
CREATE TABLE users (
    id BIGSERIAL PRIMARY KEY,
    school_id BIGINT REFERENCES schools(id),
    email VARCHAR(255) UNIQUE NOT NULL,
    phone VARCHAR(30),
    password_hash VARCHAR(255) NOT NULL,
    mfa_enabled BOOLEAN DEFAULT false,
    mfa_secret VARCHAR(255),
    status VARCHAR(20) DEFAULT 'ACTIVE' CHECK (status IN ('ACTIVE','SUSPENDED','LOCKED')),
    last_login_at TIMESTAMPTZ
);
CREATE TABLE user_roles (
    user_id BIGINT REFERENCES users(id),
    role_id BIGINT REFERENCES roles(id),
    PRIMARY KEY (user_id, role_id)
);
-- Links a user account to the domain entity it represents (staff/parent/student)
CREATE TABLE user_profiles (
    user_id BIGINT PRIMARY KEY REFERENCES users(id),
    profile_type VARCHAR(20) CHECK (profile_type IN ('STAFF','PARENT','STUDENT')),
    profile_id BIGINT NOT NULL -- FK resolved at application layer to staff.id / parents.id / students.id
);
CREATE TABLE audit_logs (
    id BIGSERIAL PRIMARY KEY,
    user_id BIGINT REFERENCES users(id),
    entity_type VARCHAR(100) NOT NULL,
    entity_id BIGINT NOT NULL,
    action VARCHAR(20) NOT NULL, -- CREATE, UPDATE, DELETE, LOGIN, EXPORT
    previous_value JSONB,
    new_value JSONB,
    ip_address VARCHAR(45),
    occurred_at TIMESTAMPTZ DEFAULT now()
);
CREATE TABLE notifications (
    id BIGSERIAL PRIMARY KEY,
    user_id BIGINT REFERENCES users(id),
    channel VARCHAR(20) CHECK (channel IN ('EMAIL','SMS','WHATSAPP','PUSH','IN_APP')),
    title VARCHAR(255),
    body TEXT,
    is_read BOOLEAN DEFAULT false,
    sent_at TIMESTAMPTZ
);
CREATE TABLE academic_years (
    id BIGSERIAL PRIMARY KEY,
    name VARCHAR(20) NOT NULL, -- '2026/2027'
    start_date DATE NOT NULL,
    end_date DATE NOT NULL,
    is_current BOOLEAN DEFAULT false
);
CREATE TABLE terms (
    id BIGSERIAL PRIMARY KEY,
    academic_year_id BIGINT REFERENCES academic_years(id),
    name VARCHAR(20) NOT NULL, -- 'Term 1'
    start_date DATE NOT NULL,
    end_date DATE NOT NULL,
    is_current BOOLEAN DEFAULT false
);
1.2 Student Tables
CREATE TABLE students (
    id BIGSERIAL PRIMARY KEY,

    first_name VARCHAR(100) NOT NULL,
    middle_name VARCHAR(100),

    transportation_required BOOLEAN DEFAULT false,

        CHECK (status IN ('PROSPECTIVE','ENROLLED','WITHDRAWN','GRADUATED','SUSPENDED'))

CREATE TABLE student_addresses (
    id BIGSERIAL PRIMARY KEY,
    student_id BIGINT REFERENCES students(id),
    address_line1 VARCHAR(255),

    city VARCHAR(100),

    country VARCHAR(100),

CREATE TABLE parents (
    id BIGSERIAL PRIMARY KEY,
    first_name VARCHAR(100) NOT NULL,
    last_name VARCHAR(100) NOT NULL,
    relationship VARCHAR(20) CHECK (relationship IN ('FATHER','MOTHER','GUARDIAN')),
    phone VARCHAR(30),
    email VARCHAR(255),
    occupation VARCHAR(100),
    is_emergency_contact BOOLEAN DEFAULT false

CREATE TABLE student_parents (
    student_id BIGINT REFERENCES students(id),
    parent_id BIGINT REFERENCES parents(id),
    is_primary_contact BOOLEAN DEFAULT false,

CREATE TABLE student_documents (
    id BIGSERIAL PRIMARY KEY,
    student_id BIGINT REFERENCES students(id),
    document_type VARCHAR(50), -- BIRTH_CERT, PREVIOUS_REPORT_CARD, MEDICAL, PHOTO, ID
    file_url VARCHAR(500),
    uploaded_by BIGINT REFERENCES users(id),
    uploaded_at TIMESTAMPTZ DEFAULT now()
);
CREATE TABLE admission_applications (
    id BIGSERIAL PRIMARY KEY,
    student_id BIGINT REFERENCES students(id),
    applied_class_id BIGINT REFERENCES classes(id),
    previous_school VARCHAR(255),
    stage VARCHAR(30) DEFAULT 'SUBMITTED'
        CHECK (stage IN ('SUBMITTED','REVIEWED','ASSESSMENT_SCHEDULED',
                          'DECISION_PENDING','OFFERED','REJECTED',
                          'FEES_PENDING','ENROLLED')),
    decision VARCHAR(20) CHECK (decision IN ('ADMIT','WAITLIST','REJECT')),
    decision_notes TEXT,
    decided_by BIGINT REFERENCES users(id),
    decided_at TIMESTAMPTZ
);
CREATE TABLE classes (
    id BIGSERIAL PRIMARY KEY,
    academic_year_id BIGINT REFERENCES academic_years(id),
    name VARCHAR(50) NOT NULL, -- 'Grade 5A'
    grade_level VARCHAR(20),
    class_teacher_id BIGINT REFERENCES staff(id),

CREATE TABLE student_enrollment (
    id BIGSERIAL PRIMARY KEY,
    student_id BIGINT REFERENCES students(id),
    class_id BIGINT REFERENCES classes(id),

    enrollment_date DATE NOT NULL,

    status VARCHAR(20) DEFAULT 'ACTIVE' CHECK (status IN ('ACTIVE','WITHDRAWN','GRADUATED'))
);

PART 2 — USER STORIES
Format: As a <role>, I want to <action>, so that <benefit>. Each includes acceptance criteria (AC).
2.1 Student Enrolment
US-01 — As an Admissions Officer, I want to record a new student application with parent/guardian details and supporting documents, so that the admission workflow can begin.
AC: Required fields enforced (name, DOB, gender, at least one parent contact); documents upload to secure storage; application status defaults to SUBMITTED.
US-02 — As a Principal, I want to review and move an application through Reviewed → Assessment → Decision → Offer stages, so that admissions follow a consistent workflow.
AC: Stage transitions are logged with timestamp and user; rejecting an application requires a reason; only forward transitions allowed (no skipping stages) unless Super Admin override.
US-03 — As an Admissions Officer, I want to convert an offered-and-paid applicant into an enrolled student with a class assignment, so that they appear in the active roster.
AC: Enrollment blocked until invoice status is PAID or WAIVED; a unique student_number is generated on enrollment; class capacity is checked before assignment.
US-04 — As a Parent, I want to view and update my child's profile information (address, emergency contact), so that the school has current records.
AC: Parent can edit only non-sensitive fields (address, phone); academic/financial fields are read-only; changes are logged in audit_logs.
2.2 Student Attendance
US-05 — As a Teacher, I want to mark daily attendance for my assigned class in under 60 seconds, so that records stay current with minimal disruption to teaching time.
AC: Default status PRESENT pre-filled for all students; teacher can bulk-toggle exceptions; submission locked after end-of-day cutoff unless admin unlocks it.
US-06 — As a Principal, I want to receive an automatic alert when a student's attendance falls below 80%, so that I can intervene early.
AC: Alert engine recalculates rolling attendance % nightly; alert appears on dashboard and is emailed to Principal + homeroom teacher; alert clears automatically once attendance recovers above threshold.
US-07 — As a Parent, I want to see my child's attendance history and receive same-day absence notifications, so that I stay informed.
AC: Notification sent within 30 minutes of an ABSENT mark with no prior excuse on file.
2.3 Staff Attendance
US-08 — As a Staff member, I want to check in/out via the portal or a biometric device, so that my hours are accurately tracked.
AC: Check-in after configured start time auto-flags LATE; duplicate check-ins on the same day are rejected.
US-09 — As an HR Officer, I want a report of staff lateness frequency over a date range, so that I can address recurring issues.
AC: Report filterable by department/staff; exportable to PDF/Excel/CSV.
2.4 Fees & Finance
US-10 — As a Finance Officer, I want to generate invoices automatically from a student's fee structure (class, boarding, transport, discounts, scholarships), so that billing is accurate and fast.
AC: Invoice generation is idempotent per student/term; applies all eligible discounts/scholarships automatically; generates a unique invoice number.
US-11 — As a Finance Officer, I want to record a payment against an invoice and issue a receipt, so that balances update in real time.
AC: Partial payments update status to PARTIAL; full payment sets PAID; receipt number is unique and printable/emailable.
US-12 — As a Proprietor, I want a real-time financial dashboard showing billed, collected, and outstanding amounts with trend charts, so that I can assess financial health at a glance.
AC: Dashboard figures refresh at least every 15 minutes; collection % calculated as amount_paid / total_amount; drill-down to debtor list.
US-13 — As a Finance Officer, I want the system to flag overdue invoices and trigger reminder notifications, so that collection improves.
AC: Invoices past due_date with balance > 0 auto-transition to OVERDUE; reminders sent at configurable intervals (e.g., 7/14/30 days overdue).
US-14 — As a Parent, I want to pay fees online and view payment history, so that I don't need to visit the school in person.
AC: Online payment integrates with a payment gateway; successful payment updates invoice within seconds via webhook; failed payments are retryable.
2.5 Academic Performance
US-15 — As a Teacher, I want to enter scores for assessments for my assigned students, so that grades are recorded centrally.
AC: Score entry validated against max_score; grade letter auto-computed from a configurable grading scale; entry locked after term closes unless admin unlocks.
US-16 — As a Vice Principal, I want to view class and subject performance analytics per term, so that I can identify underperforming areas.
AC: Analytics include average, pass rate, and term-over-term trend; filterable by class/subject/teacher.
US-17 — As the system, I want to automatically flag students as "at risk" based on repeated failure, grade decline, or low attendance + poor performance, so that staff can intervene proactively.
AC: Flag logic runs after each term's grades are finalized; flags visible on Principal and homeroom teacher dashboards; flags can be manually resolved with notes.
US-18 — As a Parent, I want to view and download my child's report card once published, so that I can track academic progress.
AC: Report card only visible to parent after status = PUBLISHED; downloadable as PDF.
2.6 Curriculum Coverage
US-19 — As a Vice Principal, I want to set up the term's curriculum topics per subject/class, so that teachers have a clear teaching plan.
AC: Topics have a planned week and sequence order; bulk-import via CSV supported.
US-20 — As a Teacher, I want to log which curriculum topics I've taught and their completion status, so that coverage is tracked accurately.
AC: Teacher selects topic, date taught, and completion status; incomplete topics require a reason.
US-21 — As a Principal, I want a curriculum coverage dashboard comparing planned vs. actual progress by teacher/subject, so that I can address delays.
AC: Dashboard shows % completion and flags subjects behind schedule (configurable threshold, e.g., >2 weeks behind).
2.7 Student Discipline
US-22 — As a Teacher, I want to report a discipline incident with details and category, so that it enters the review workflow.
AC: Incident requires student, category, description, date; status defaults to REPORTED.
US-23 — As a Vice Principal, I want to move an incident through Review → Investigation → Decision → Action → Follow-up, so that discipline is handled consistently.
AC: Parent notification is logged with timestamp; action taken is selected from a controlled list; follow-up notes are required to close the incident.
US-24 — As a Principal, I want a report of repeated discipline incidents per student, so that I can identify patterns requiring deeper intervention.
AC: Report is filterable by incident type and date range; triggers an alert after N incidents within a rolling period.
2.8 Parent Experience
US-25 — As a Parent, I want to submit a complaint through the portal or mobile app with a category, so that my concern is tracked to resolution.
AC: Complaint gets a tracking ID; parent receives acknowledgment notification within a configurable SLA (e.g., 24 hours).
US-26 — As a Finance/Academic/Facilities staff member, I want complaints in my category auto-routed to me, so that response times improve.
AC: Routing based on category; reassignment possible by Principal; SLA breach triggers an alert.
US-27 — As a Parent, I want to complete a periodic satisfaction survey, so that the school can act on my feedback.
AC: Survey covers configured categories (teaching quality, communication, safety, etc.); results aggregate into a satisfaction score visible on the executive dashboard.
2.9 HR / Staffing
US-28 — As an HR Officer, I want to post a job vacancy and track applicants through interview to hire, so that recruitment is centrally managed.
AC: Applicant stage changes are logged; hiring an applicant can auto-create a staff record.
US-29 — As a Staff member, I want to submit a leave request and track its approval status, so that I know when I'm covered.
AC: Request routes to the staff member's supervisor; approved leave auto-marks staff_attendance.status = LEAVE for the date range.
US-30 — As an HR Officer, I want to record performance review scores per staff member per year, so that performance trends are visible over time.
AC: Review pulls attendance % and curriculum completion % automatically where applicable; overall score is a weighted composite (weights configurable).
2.10 Facilities
US-31 — As any staff member, I want to submit a maintenance request for a facility or asset, so that issues get resolved.
AC: Request requires priority; urgent requests trigger an immediate alert to the Facilities Manager.
US-32 — As a Facilities Manager, I want to assign, progress, and close maintenance requests with cost tracking, so that repair spend is visible.
AC: Closing a request requires a recorded cost (can be 0) and creates a maintenance_history entry.
2.11 Executive Intelligence
US-33 — As a Proprietor, I want a single executive dashboard with KPIs across students, finance, academics, staff, parents, and facilities, so that I get a holistic view without visiting each module.
AC: Each KPI card links through to its detailed module view; dashboard respects role-based visibility (Proprietor sees summaries, not confidential detail, unless authorized).
US-34 — As a Principal, I want to receive prioritized alerts (financial, academic, attendance, curriculum, discipline, parent, facilities) in one place, so that I can act on the most urgent issues first.
AC: Alerts are sortable/filterable by category and severity; each alert deep-links to the relevant record; alerts can be marked acknowledged/resolved.
PART 3 — REST API STRUCTURE
3.0 Conventions
Base URL: https://api.smpis.app/v1
Auth: Authorization: Bearer <JWT>; JWT encodes user_id, school_id, roles[].
Standard response envelope:
{
  "data": { },
  "meta": { "page": 1, "per_page": 25, "total": 130 },
  "errors": []
}
Standard error shape: { "errors": [{ "code": "VALIDATION_ERROR", "field": "date_of_birth", "message": "..." }] }
Pagination via ?page=&per_page=; filtering via ?filter[status]=; sorting via ?sort=-created_at.
All write endpoints require role-based permission (permission.code from permissions table); enforced by middleware, not per-endpoint logic.

APPENDIX — Traceability to SRS
Every module in SRS §3 maps 1:1 to a schema group (Part 1), a user-story cluster (Part 2), an API resource group (Part 3), and a wireframe set (Part 4).
Role-based access (SRS §4) is implemented via roles / permissions / role_permissions / user_roles, enforced at the API middleware layer — not hardcoded per screen.
Alert Engine (SRS §16) is implemented as a scheduled job layer reading from the operational tables and writing to a lightweight alerts table (recommended addition: alerts(id, category, severity, message, entity_type, entity_id, status, created_at)), surfaced via /alerts.
Recommended next steps per SRS §30: proceed to detailed PRD sign-off, confirm grading-scale and alert-threshold business rules with school leadership (these are configuration, not hardcoded), and begin Phase 1 sprint planning against the module list in SRS §26.
